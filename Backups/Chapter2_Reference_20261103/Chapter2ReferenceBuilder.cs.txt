  var saved=root;var group=new GameObject(name).transform;group.SetParent(root);root=group;
  Beam("OldTree_Trunk",p,p+new Vector3(-.5f,6,0),1.1f*scale,timber);
  var v=new List<Vector3>();var tr=new List<int>();UnityEngine.Random.InitState(317);
  for(int branch=0;branch<10;branch++){
   float a=branch*2.399f;var end=p+new Vector3(Mathf.Cos(a)*4.7f,7.5f+(branch%3)*.8f,Mathf.Sin(a)*4.2f)*scale;
   Beam("OldTree_Branch",p+Vector3.up*4*scale,end,.28f*scale,timber);
   for(int n=0;n<65;n++){
    var q=end+UnityEngine.Random.insideUnitSphere*1.8f*scale;var normal=UnityEngine.Random.onUnitSphere;var right=Vector3.Cross(normal,Vector3.up).normalized;if(right.sqrMagnitude<.1f)right=Vector3.right;var up=Vector3.Cross(normal,right);float r=UnityEngine.Random.Range(.09f,.18f)*scale;
    for(int petal=0;petal<5;petal++){float ang=petal*Mathf.PI*2/5;var d=right*Mathf.Cos(ang)+up*Mathf.Sin(ang);var tangent=Vector3.Cross(normal,d);int b=v.Count;v.Add(q);v.Add(q+d*r+tangent*r*.42f);v.Add(q+d*r*1.65f);v.Add(q+d*r-tangent*r*.42f);tr.AddRange(new[]{b,b+1,b+2,b,b+2,b+3});}
   }
  }MeshObject(name+"_650_FivePetalBlossoms",v,tr,pink);root=saved;
 }

 }

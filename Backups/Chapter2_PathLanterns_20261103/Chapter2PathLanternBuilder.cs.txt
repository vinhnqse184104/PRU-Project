using System;
using System.IO;
using System.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
public static class Chapter2PathLanternBuilder {
 const string A="Assets/Chapter2PathLanterns";
 static Material wood,metal,glow;static Transform author;
 static GameObject Piece(string name,PrimitiveType type,Vector3 p,Vector3 scale,Material mat){var g=GameObject.CreatePrimitive(type);g.name=name;g.transform.SetParent(author,false);g.transform.localPosition=p;g.transform.localScale=scale;UnityEngine.Object.DestroyImmediate(g.GetComponent<Collider>());g.GetComponent<Renderer>().sharedMaterial=mat;g.layer=2;GameObjectUtility.SetStaticEditorFlags(g,StaticEditorFlags.BatchingStatic);return g;}
 static Material Material(string name,Color color){var m=new Material(Shader.Find("Universal Render Pipeline/Lit"));m.name=name;m.SetColor("_BaseColor",color);m.SetFloat("_Smoothness",.18f);m.enableInstancing=true;AssetDatabase.CreateAsset(m,A+"/"+name+".mat");return m;}
 static void MakeLantern(bool post){
  if(post){Piece("TimberPost",PrimitiveType.Cylinder,new Vector3(0,1.16f,0),new Vector3(.15f,1.16f,.15f),wood);Piece("CrossArm",PrimitiveType.Cube,new Vector3(.28f,2.28f,0),new Vector3(.72f,.10f,.11f),wood);var brace=Piece("DiagonalBrace",PrimitiveType.Cylinder,new Vector3(.16f,2.05f,0),new Vector3(.055f,.23f,.055f),wood);brace.transform.localRotation=Quaternion.Euler(0,0,-40);}
  var offset=post?new Vector3(.53f,1.91f,0):Vector3.zero;
  Piece("HangingLink",PrimitiveType.Cylinder,offset+Vector3.up*.26f,new Vector3(.035f,.12f,.035f),metal);
  Piece("AmberLanternGlass",PrimitiveType.Cube,offset,new Vector3(.27f,.37f,.27f),glow);
  for(int x=-1;x<=1;x+=2)for(int z=-1;z<=1;z+=2)Piece("WoodenCornerFrame",PrimitiveType.Cube,offset+new Vector3(x*.155f,0,z*.155f),new Vector3(.037f,.43f,.037f),wood);
  foreach(float y in new[]{-.215f,.215f})Piece("LanternWoodRim",PrimitiveType.Cube,offset+Vector3.up*y,new Vector3(.37f,.055f,.37f),wood);
  var mesh=new Mesh();mesh.name=post?"PostLanternRoof":"PorchLanternRoof";mesh.vertices=new[]{new Vector3(-.26f,.23f,-.26f),new Vector3(.26f,.23f,-.26f),new Vector3(.26f,.23f,.26f),new Vector3(-.26f,.23f,.26f),new Vector3(0,.40f,0)};mesh.triangles=new[]{0,4,1,1,4,2,2,4,3,3,4,0,0,1,2,0,2,3};mesh.RecalculateNormals();AssetDatabase.CreateAsset(mesh,A+"/"+mesh.name+".asset");var roof=new GameObject("SmallPitchedRoof");roof.transform.SetParent(author,false);roof.transform.localPosition=offset;roof.AddComponent<MeshFilter>().sharedMesh=mesh;roof.AddComponent<MeshRenderer>().sharedMaterial=wood;roof.layer=2;
  var obj=new GameObject("WarmPointLight");obj.transform.SetParent(author,false);obj.transform.localPosition=offset;var light=obj.AddComponent<Light>();light.type=LightType.Point;light.color=new Color(1,.60f,.22f);light.intensity=3.2f;light.range=6.5f;light.shadows=LightShadows.None;light.renderMode=LightRenderMode.Auto;
 }
 public static void Build(){
  if(EditorApplication.isPlaying)throw new Exception("Stop Play before authoring");if(AssetDatabase.IsValidFolder(A)){var mat=AssetDatabase.LoadAssetAtPath<Material>(A+"/LanternAmberEmission.mat");mat.shader=Shader.Find("Chapter2/LanternEmission");mat.SetColor("_EmissionColor",new Color(2.8f,1.12f,.182f,1));EditorUtility.SetDirty(mat);AssetDatabase.SaveAssets();var gv=EditorWindow.GetWindow(Type.GetType("UnityEditor.GameView,UnityEditor"));gv.Show();gv.Focus();gv.maximized=true;return;}var original=SceneManager.GetActiveScene();if(original.path!=Chapter2EnvironmentTool.ScenePath||original.isDirty)throw new Exception("Open saved Chapter2 first");if(original.GetRootGameObjects().Any(x=>x.name=="Chapter2_Path_Lanterns"))throw new Exception("Path lanterns already exist; refusing duplicate placement");
  var terrain=UnityEngine.Object.FindAnyObjectByType<Terrain>();AssetDatabase.CreateFolder("Assets","Chapter2PathLanterns");wood=Material("LanternWeatheredTimber",new Color(.30f,.17f,.075f));var old=AssetDatabase.LoadAssetAtPath<Material>("Assets/Chapter2ReferenceRedesign/OldDarkTimber.mat");if(old&&old.GetTexture("_BaseMap"))wood.SetTexture("_BaseMap",old.GetTexture("_BaseMap"));metal=Material("LanternIron",new Color(.09f,.075f,.055f));glow=Material("LanternAmberEmission",new Color(1,.55f,.12f));glow.EnableKeyword("_EMISSION");glow.SetColor("_EmissionColor",new Color(1,.40f,.065f)*2.8f);
  var staging=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);SceneManager.SetActiveScene(staging);author=new GameObject("VillagePathLantern_2_4m").transform;MakeLantern(true);var post=PrefabUtility.SaveAsPrefabAsset(author.gameObject,A+"/VillagePathLantern.prefab");UnityEngine.Object.DestroyImmediate(author.gameObject);author=new GameObject("VillagePorchLantern").transform;MakeLantern(false);var porch=PrefabUtility.SaveAsPrefabAsset(author.gameObject,A+"/VillagePorchLantern.prefab");UnityEngine.Object.DestroyImmediate(author.gameObject);
  var group=new GameObject("Chapter2_Path_Lanterns");var start=new Vector3(97.64f,0,89.05f);var end=new Vector3(81.98f,0,31.29f);var dir=(end-start).normalized;var side=Vector3.Cross(Vector3.up,dir);int count=0;var log=new System.Text.StringBuilder();
  for(int i=0;i<11;i++)for(int sign=-1;sign<=1;sign+=2){float distance=4.6f+i*5.15f+(sign>0?.7f:0)+(i%3-1)*.20f;var p=start+dir*distance+side*sign*(3.25f+(i%2)*.15f);p.y=terrain.SampleHeight(p)+terrain.transform.position.y;var go=(GameObject)PrefabUtility.InstantiatePrefab(post,group.transform);go.name="PathLantern_"+(sign<0?"Left_":"Right_")+(i+1).ToString("00");go.transform.position=p;go.transform.rotation=Quaternion.LookRotation(Vector3.Cross(-side*sign,Vector3.up),Vector3.up);go.transform.localScale=Vector3.one*(.98f+(i%3)*.015f);count++;log.AppendLine(go.name+" position="+p.ToString("F3")+" routeDistance="+distance+" corridorClearance>=2.6m");}
  foreach(var p in new[]{new Vector3(79.1f,3.05f,31.35f),new Vector3(86.65f,3.05f,31.35f)}){var go=(GameObject)PrefabUtility.InstantiatePrefab(porch,group.transform);go.name="Porch_HangingLantern";go.transform.position=p;}
  AssetDatabase.SaveAssets();EditorSceneManager.SaveScene(staging,A+"/LanternsOnly.unity");log.AppendLine("Posts="+count+" Porch=2 NewColliders="+group.GetComponentsInChildren<Collider>(true).Length+" ShadowCastingLights="+group.GetComponentsInChildren<Light>(true).Count(l=>l.shadows!=LightShadows.None));File.WriteAllText(Chapter2EnvironmentTool.Output+"lantern_inventory.txt",log.ToString());EditorSceneManager.CloseScene(staging,true);SceneManager.SetActiveScene(original);File.WriteAllText(Chapter2EnvironmentTool.Output+"build.done","24 real prefab instances staged");
 }
 public static void Preview(){var go=new GameObject("Temporary_PathReviewCamera");var c=go.AddComponent<Camera>();go.transform.position=new Vector3(91,3.2f,67);go.transform.LookAt(new Vector3(83.5f,1.8f,36));c.fieldOfView=65;c.farClipPlane=300;Chapter2EnvironmentTool.Capture(c,"Path_preview");UnityEngine.Object.DestroyImmediate(go);}
}
